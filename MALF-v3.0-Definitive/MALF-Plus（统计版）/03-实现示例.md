```python
from dataclasses import dataclass, field
from typing import Literal, List, Dict, Tuple
from datetime import datetime
import bisect

# ==================== 数据结构 ====================

@dataclass
class PriceBar:
    """输入：K线"""
    symbol: str
    timestamp: datetime
    timeframe: str  # Plus需要
    open: float
    high: float
    low: float
    close: float

@dataclass
class WavePosition:
    """输出：波段位置（6字段）"""
    direction: Literal["UP", "DOWN"]
    new_count: int
    no_new_span: int
    life_state: Literal["alive", "broken", "reborn"]
    update_rank: float | None = None
    stagnation_rank: float | None = None

@dataclass
class WaveState:
    """内部状态"""
    symbol: str
    timeframe: str
    current_high: float
    current_low: float
    guard_high: float | None = None
    guard_low: float | None = None
    new_count: int = 0
    no_new_span: int = 0
    life_state: str = "alive"
    direction: Literal["UP", "DOWN"] | None = None

@dataclass
class WaveSample:
    """单个样本"""
    new_count: int
    no_new_span: int
    end_time: datetime
    end_price: float

@dataclass
class SamplePool:
    """样本池"""
    symbol: str
    timeframe: str
    direction: str
    samples: List[WaveSample] = field(default_factory=list)
    new_count_sorted: List[int] = field(default_factory=list)
    no_new_span_sorted: List[int] = field(default_factory=list)
    
    def add_sample(self, sample: WaveSample) -> None:
        """添加样本并保持排序"""
        self.samples.append(sample)
        bisect.insort(self.new_count_sorted, sample.new_count)
        bisect.insort(self.no_new_span_sorted, sample.no_new_span)

# ==================== 样本池管理器 ====================

class PoolManager:
    """样本池管理器"""
    
    def __init__(self):
        # 三维索引: (symbol, timeframe, direction) -> SamplePool
        self.pools: Dict[Tuple[str, str, str], SamplePool] = {}
    
    def get_pool(self, symbol: str, timeframe: str, 
                 direction: str) -> SamplePool:
        """获取样本池"""
        key = (symbol, timeframe, direction)
        if key not in self.pools:
            self.pools[key] = SamplePool(
                symbol=symbol,
                timeframe=timeframe,
                direction=direction
            )
        return self.pools[key]
    
    def add_sample(self, symbol: str, timeframe: str, 
                   direction: str, sample: WaveSample) -> None:
        """添加样本"""
        pool = self.get_pool(symbol, timeframe, direction)
        pool.add_sample(sample)
    
    def calc_rank(self, symbol: str, timeframe: str, 
                  direction: str, new_count: int, 
                  no_new_span: int) -> Tuple[float | None, float | None]:
        """计算分位"""
        pool = self.get_pool(symbol, timeframe, direction)
        
        # 样本门槛
        if len(pool.samples) < 30:
            return None, None
        
        # 二分查找
        pos1 = bisect.bisect_left(pool.new_count_sorted, new_count)
        update_rank = pos1 / len(pool.samples)
        
        pos2 = bisect.bisect_left(pool.no_new_span_sorted, no_new_span)
        stag_rank = pos2 / len(pool.samples)
        
        return update_rank, stag_rank

# ==================== 核心引擎 ====================

class MALFPlus:
    """MALF-Plus核心引擎"""
    
    def __init__(self):
        self.states: Dict[Tuple[str, str], WaveState] = {}
        self.pool_manager = PoolManager()
    
    def on_bar(self, bar: PriceBar) -> WavePosition:
        """每根K线调用"""
        
        key = (bar.symbol, bar.timeframe)
        
        if key not in self.states:
            # 初始化
            self.states[key] = self._init_state(bar)
        
        state = self.states[key]
        
        # 1. Mini处理（前4字段）
        推进标志 = self._check_progress(state, bar)
        
        if not 推进标志:
            self._check_break(state, bar)
        
        self._check_reborn(state, bar)
        
        if not 推进标志 and state.life_state == "alive":
            state.no_new_span += 1
        
        # 2. 检查入池
        if state.life_state == "broken":
            self._add_to_pool(state, bar)
        
        # 3. 计算历史位置（后2字段）
        if state.life_state == "alive":
            update_rank, stag_rank = self.pool_manager.calc_rank(
                state.symbol, state.timeframe, state.direction,
                state.new_count, state.no_new_span
            )
        else:
            update_rank, stag_rank = None, None
        
        # 4. 输出
        return WavePosition(
            direction=state.direction or "UP",
            new_count=state.new_count,
            no_new_span=state.no_new_span,
            life_state=state.life_state,
            update_rank=update_rank,
            stagnation_rank=stag_rank
        )
    
    def _init_state(self, bar: PriceBar) -> WaveState:
        """初始化"""
        return WaveState(
            symbol=bar.symbol,
            timeframe=bar.timeframe,
            current_high=bar.high,
            current_low=bar.low,
            direction="UP"
        )
    
    def _check_progress(self, state: WaveState, bar: PriceBar) -> bool:
        """检查推进"""
        if state.direction == "UP":
            if bar.high > state.current_high:
                state.current_high = bar.high
                state.new_count += 1
                state.no_new_span = 0
                return True
        elif state.direction == "DOWN":
            if bar.low < state.current_low:
                state.current_low = bar.low
                state.new_count += 1
                state.no_new_span = 0
                return True
        return False
    
    def _check_break(self, state: WaveState, bar: PriceBar) -> None:
        """检查守护破坏"""
        if state.life_state != "alive":
            return
        
        if state.direction == "UP":
            if state.guard_low and bar.low < state.guard_low:
                state.life_state = "broken"
        elif state.direction == "DOWN":
            if state.guard_high and bar.high > state.guard_high:
                state.life_state = "broken"
    
    def _check_reborn(self, state: WaveState, bar: PriceBar) -> None:
        """检查重生"""
        if state.life_state != "broken":
            return
        
        if bar.high > state.current_high:
            state.direction = "UP"
            state.life_state = "alive"
            state.new_count = 1
            state.no_new_span = 0
            state.current_high = bar.high
            state.guard_low = bar.low
        elif bar.low < state.current_low:
            state.direction = "DOWN"
            state.life_state = "alive"
            state.new_count = 1
            state.no_new_span = 0
            state.current_low = bar.low
            state.guard_high = bar.high
    
    def _add_to_pool(self, state: WaveState, bar: PriceBar) -> None:
        """入池"""
        sample = WaveSample(
            new_count=state.new_count,
            no_new_span=state.no_new_span,
            end_time=bar.timestamp,
            end_price=state.current_high if state.direction == "UP" 
                     else state.current_low
        )
        
        self.pool_manager.add_sample(
            state.symbol, state.timeframe, state.direction, sample
        )

# ==================== 使用示例 ====================

if __name__ == "__main__":
    engine = MALFPlus()
    
    # 模拟数据
    bars = [
        PriceBar("000001", datetime.now(), "日线", 100, 105, 95, 103),
        PriceBar("000001", datetime.now(), "日线", 103, 110, 100, 108),
        PriceBar("000001", datetime.now(), "日线", 108, 112, 105, 110),
    ]
    
    for bar in bars:
        wp = engine.on_bar(bar)
        print(f"HH={bar.high}, LL={bar.low}")
        print(f"  → {wp}")
        
        # 显示历史位置
        if wp.update_rank is not None:
            print(f"  → 更新位置: 前{(1-wp.update_rank)*100:.0f}%")
        if wp.stagnation_rank is not None:
            print(f"  → 停滞位置: 前{(1-wp.stagnation_rank)*100:.0f}%")
```

## 🔧 持久化示例（SQLite）

```python
import sqlite3

class PersistentPoolManager(PoolManager):
    """带持久化的样本池管理器"""
    
    def __init__(self, db_path: str):
        super().__init__()
        self.db = sqlite3.connect(db_path)
        self._init_db()
        self._load_all()
    
    def _init_db(self):
        """初始化数据库"""
        self.db.execute("""
            CREATE TABLE IF NOT EXISTS sample_pool (
                symbol TEXT NOT NULL,
                timeframe TEXT NOT NULL,
                direction TEXT NOT NULL,
                new_count INTEGER NOT NULL,
                no_new_span INTEGER NOT NULL,
                end_time TEXT NOT NULL,
                end_price REAL NOT NULL,
                PRIMARY KEY (symbol, timeframe, direction, end_time)
            )
        """)
        self.db.commit()
    
    def _load_all(self):
        """加载所有样本"""
        cursor = self.db.execute("""
            SELECT symbol, timeframe, direction, 
                   new_count, no_new_span, end_time, end_price
            FROM sample_pool
        """)
        
        for row in cursor:
            sample = WaveSample(
                new_count=row[3],
                no_new_span=row[4],
                end_time=datetime.fromisoformat(row[5]),
                end_price=row[6]
            )
            super().add_sample(row[0], row[1], row[2], sample)
    
    def add_sample(self, symbol: str, timeframe: str,
                   direction: str, sample: WaveSample) -> None:
        """添加样本（持久化）"""
        # 内存
        super().add_sample(symbol, timeframe, direction, sample)
        
        # 数据库
        self.db.execute("""
            INSERT OR REPLACE INTO sample_pool
            VALUES (?, ?, ?, ?, ?, ?, ?)
        """, (symbol, timeframe, direction, 
              sample.new_count, sample.no_new_span,
              sample.end_time.isoformat(), sample.end_price))
        self.db.commit()

# 使用
engine = MALFPlus()
engine.pool_manager = PersistentPoolManager("malf_plus.db")
```

### 应用1: 信号过滤

```python
def should_trade(wp: WavePosition) -> bool:
    """根据历史位置过滤信号"""
    
    if wp.update_rank is None:
        return True  # 样本不足，不过滤
    
    # 只交易强势波段
    if wp.update_rank > 0.6 and wp.stagnation_rank < 0.4:
        return True  # 创新多，停滞少
    
    # 过滤老化波段
    if wp.update_rank > 0.8 and wp.stagnation_rank > 0.7:
        return False  # 已老化
    
    return True
```

### 应用2: 仓位调整

```python
def calc_position_size(wp: WavePosition, 
                       base_size: float) -> float:
    """根据历史位置调整仓位"""
    
    if wp.update_rank is None:
        return base_size
    
    # 强势波段 → 加大仓位
    if wp.update_rank > 0.7 and wp.stagnation_rank < 0.3:
        return base_size * 1.5
    
    # 老化波段 → 减小仓位
    if wp.update_rank > 0.8 or wp.stagnation_rank > 0.7:
        return base_size * 0.5
    
    return base_size
```
