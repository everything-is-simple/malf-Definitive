```python
    life_state: str = "alive"
```

### 样本池结构

```python
@dataclass
class WaveSample:
    """单个样本"""
    new_count: int
    no_new_span: int
    终止时间: datetime
    终止价格: float

@dataclass  
class SamplePool:
    """样本池"""
    symbol: str
    timeframe: str
    direction: str
    samples: List[WaveSample]
    
    # 优化：预排序列表
    new_count_sorted: List[int] = field(default_factory=list)
    no_new_span_sorted: List[int] = field(default_factory=list)
    
    def add_sample(self, sample: WaveSample) -> None:
        """添加样本并保持排序"""
        self.samples.append(sample)
        
        # 插入排序（保持有序）
        import bisect
        bisect.insort(self.new_count_sorted, sample.new_count)
        bisect.insort(self.no_new_span_sorted, sample.no_new_span)
```

### 增强输出

```python
@dataclass
class WavePosition:
    """Plus版输出（6字段）"""
    
    # Mini的4个字段
    direction: Literal["UP", "DOWN"]
    new_count: int
    no_new_span: int
    life_state: Literal["alive", "broken", "reborn"]
    
    # Plus新增的2个字段
    update_rank: float | None       # [0, 1) 或 None
    stagnation_rank: float | None   # [0, 1) 或 None
    
    def __str__(self) -> str:
        rank_str = ""
        if self.update_rank is not None:
            rank_str += f", 更新位置{self.update_rank:.1%}"
        if self.stagnation_rank is not None:
            rank_str += f", 停滞位置{self.stagnation_rank:.1%}"
        
        return f"WavePosition({self.direction}, " \
               f"推进{self.new_count}次, " \
               f"停滞{self.no_new_span}期, " \
               f"{self.life_state}{rank_str})"
```

## 🔧 核心算法

### 主循环（on_bar）

```python
def on_bar(state: WaveState, 
           bar: PriceBar,
           pool_manager: PoolManager) -> WavePosition:
    """
    每根K线调用一次（Plus版）
    
    Args:
        state: 当前状态（Mini）
        bar: 当前K线
        pool_manager: 样本池管理器
        
    Returns:
        WavePosition: 6字段
    """
    
    # 1. Mini的处理（前4字段）
    mini_wp = mini_on_bar(state, bar)
    
    # 2. 检查是否需要入池
    if mini_wp.life_state == "broken":
        入池(state, pool_manager)
    
    # 3. 计算历史位置（后2字段）
    if mini_wp.life_state == "alive":
        update_rank, stag_rank = 计算分位(
            state, pool_manager
        )
    else:
        update_rank, stag_rank = None, None
    
    # 4. 输出
    return WavePosition(
        direction=mini_wp.direction,
        new_count=mini_wp.new_count,
        no_new_span=mini_wp.no_new_span,
        life_state=mini_wp.life_state,
        update_rank=update_rank,
        stagnation_rank=stag_rank
    )
```

### 算法1: 样本池管理

```python
class PoolManager:
    """样本池管理器"""
    
    def __init__(self):
        # 三维索引
        self.pools: Dict[Tuple[str, str, str], SamplePool] = {}
    
    def get_pool(self, symbol: str, 
                 timeframe: str, 
                 direction: str) -> SamplePool:
        """获取样本池（三维key）"""
        key = (symbol, timeframe, direction)
        
        if key not in self.pools:
            self.pools[key] = SamplePool(
                symbol=symbol,
                timeframe=timeframe,
                direction=direction,
                samples=[]
            )
        
        return self.pools[key]
    
    def add_sample(self, symbol: str,
                   timeframe: str,
                   direction: str,
                   sample: WaveSample) -> None:
        """添加样本"""
        pool = self.get_pool(symbol, timeframe, direction)
        pool.add_sample(sample)

def 入池(state: WaveState, 
        pool_manager: PoolManager) -> None:
    """波段终止时入池"""
    
    sample = WaveSample(
        new_count=state.new_count,
        no_new_span=state.no_new_span,
        终止时间=datetime.now(),
        终止价格=state.current_high if state.direction == "UP" 
                 else state.current_low
    )
    
    pool_manager.add_sample(
        symbol=state.symbol,
        timeframe=state.timeframe,
        direction=state.direction,
        sample=sample
    )
```

### 算法2: 分位计算（优化版）

```python
def 计算分位(state: WaveState,
           pool_manager: PoolManager) -> Tuple[float | None, float | None]:
    """
    计算历史位置（优化：二分查找）
    
    Returns:
        (update_rank, stagnation_rank)
    """
    
    pool = pool_manager.get_pool(
        state.symbol, 
        state.timeframe, 
        state.direction
    )
    
    # 样本门槛
    if len(pool.samples) < 30:
        return None, None
    
    # 二分查找（O(logN)）
    import bisect
    
    # update_rank
    pos1 = bisect.bisect_left(
        pool.new_count_sorted, 
        state.new_count
    )
    update_rank = pos1 / len(pool.samples)
    
    # stagnation_rank
    pos2 = bisect.bisect_left(
        pool.no_new_span_sorted, 
        state.no_new_span
    )
    stag_rank = pos2 / len(pool.samples)
    
    return update_rank, stag_rank
```

**复杂度**: O(logN)  
**优化**: 使用预排序列表 + 二分查找

### 算法3: 分位计算（朴素版）

```python
def 计算分位_朴素(当前值: int, 
                历史列表: List[int]) -> float | None:
    """
    朴素算法（教学用）
    
    复杂度: O(N)
    """
    
    if len(历史列表) < 30:
        return None
    
    # 严格小于
    小于当前的数量 = sum(1 for v in 历史列表 if v < 当前值)
    
    return 小于当前的数量 / len(历史列表)
```

**读取**:
```python
def load_pool(db, symbol, timeframe, direction) -> SamplePool:
    cursor = db.execute("""
        SELECT new_count, no_new_span 
        FROM sample_pool 
        WHERE symbol=? AND timeframe=? AND direction=?
        ORDER BY end_time
    """, (symbol, timeframe, direction))
    
    samples = [WaveSample(nc, nns, ...) for nc, nns in cursor]
    return SamplePool(..., samples=samples)
```

**写入**:
```python
def save_sample(db, symbol, timeframe, direction, sample):
    db.execute("""
        INSERT INTO sample_pool 
        VALUES (?, ?, ?, ?, ?, ?, ?)
    """, (symbol, timeframe, direction, 
          sample.new_count, sample.no_new_span, 
          sample.终止时间, sample.终止价格))
    db.commit()
```

```python
def 清理旧样本(pool, 保留天数=365):
    """删除超过1年的样本"""
    cutoff = datetime.now() - timedelta(days=保留天数)
```
