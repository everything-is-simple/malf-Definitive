```python
from dataclasses import dataclass
from typing import Literal
from datetime import datetime

# ==================== 数据结构 ====================

@dataclass
class PriceBar:
    """输入：K线"""
    symbol: str
    timestamp: datetime
    open: float
    high: float
    low: float
    close: float

@dataclass  
class WavePosition:
    """输出：波段位置"""
    direction: Literal["UP", "DOWN"]
    new_count: int
    no_new_span: int
    life_state: Literal["alive", "broken", "reborn"]

@dataclass
class WaveState:
    """内部状态"""
    current_high: float
    current_low: float
    guard_high: float | None = None
    guard_low: float | None = None
    new_count: int = 0
    no_new_span: int = 0
    life_state: str = "alive"
    direction: Literal["UP", "DOWN"] | None = None

# ==================== 核心引擎 ====================

class MALFMini:
    """MALF-Mini核心引擎"""
    
    def __init__(self):
        self.state: WaveState | None = None
    
    def on_bar(self, bar: PriceBar) -> WavePosition:
        """每根K线调用"""
        
        if self.state is None:
            # 初始化
            self.state = self._init_state(bar)
            return self._output()
        
        # 1. 检查推进
        推进标志 = self._check_progress(bar)
        
        # 2. 检查守护破坏
        if not 推进标志:
            self._check_break(bar)
        
        # 3. 检查重生
        self._check_reborn(bar)
        
        # 4. 更新停滞
        if not 推进标志 and self.state.life_state == "alive":
            self.state.no_new_span += 1
        
        return self._output()
    
    def _init_state(self, bar: PriceBar) -> WaveState:
        """初始化"""
        return WaveState(
            current_high=bar.high,
            current_low=bar.low,
            direction="UP"  # 简化：默认向上
        )
    
    def _check_progress(self, bar: PriceBar) -> bool:
        """检查推进（HH/LL）"""
        if self.state.direction == "UP":
            if bar.high > self.state.current_high:
                # HH
                self.state.current_high = bar.high
                self.state.new_count += 1
                self.state.no_new_span = 0
                return True
        elif self.state.direction == "DOWN":
            if bar.low < self.state.current_low:
                # LL
                self.state.current_low = bar.low
                self.state.new_count += 1
                self.state.no_new_span = 0
                return True
        return False
    
    def _check_break(self, bar: PriceBar) -> None:
        """检查守护破坏"""
        if self.state.life_state != "alive":
            return
        
        if self.state.direction == "UP":
            if self.state.guard_low and bar.low < self.state.guard_low:
                self.state.life_state = "broken"
        elif self.state.direction == "DOWN":
            if self.state.guard_high and bar.high > self.state.guard_high:
                self.state.life_state = "broken"
    
    def _check_reborn(self, bar: PriceBar) -> None:
        """检查重生"""
        if self.state.life_state != "broken":
            return
        
        # 简化：检查新极值
        if bar.high > self.state.current_high:
            # HH → 确认上升
            self.state.direction = "UP"
            self.state.life_state = "alive"
            self.state.new_count = 1
            self.state.no_new_span = 0
            self.state.current_high = bar.high
            self.state.guard_low = bar.low
        elif bar.low < self.state.current_low:
            # LL → 确认下降
            self.state.direction = "DOWN"
            self.state.life_state = "alive"
            self.state.new_count = 1
            self.state.no_new_span = 0
            self.state.current_low = bar.low
            self.state.guard_high = bar.high
    
    def _output(self) -> WavePosition:
        """输出"""
        return WavePosition(
            direction=self.state.direction or "UP",
            new_count=self.state.new_count,
            no_new_span=self.state.no_new_span,
            life_state=self.state.life_state
        )

# ==================== 使用示例 ====================

if __name__ == "__main__":
    engine = MALFMini()
    
    # 模拟数据
    bars = [
        PriceBar("TEST", datetime.now(), 100, 105, 95, 103),
        PriceBar("TEST", datetime.now(), 103, 110, 100, 108),
        PriceBar("TEST", datetime.now(), 108, 112, 105, 110),
    ]
    
    for bar in bars:
        wp = engine.on_bar(bar)
        print(f"HH={bar.high}, LL={bar.low} → {wp}")
```
